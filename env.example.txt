# URL pública do CDN R2 (sem barra no final)
# Exemplo: https://cdn.seudominio.com
R2_CDN_URL=https://cdn.seudominio.com

# Endpoint S3 do R2
# Formato: https://<account-id>.r2.cloudflarestorage.com
# Encontre o Account ID no dashboard do Cloudflare R2
S3_ENDPOINT=https://seu-account-id.r2.cloudflarestorage.com

# Região (use "auto" para R2)
R2_REGION=auto

# Access Key ID do R2
# Obtenha em: Cloudflare Dashboard → R2 → Settings → R2 API Tokens
R2_ACCESS_KEY_ID=seu_access_key_id_aqui

# Secret Access Key do R2
# Obtenha junto com o Access Key ID (salve em local seguro!)
R2_SECRET_ACCESS_KEY=seu_secret_access_key_aqui

# Nome do bucket R2
# O nome do bucket que você criou no Cloudflare R2
R2_BUCKET=ot-launcher-files

# Caminho para a pasta do cliente (caminho absoluto)
# Windows: C:\Users\Usuario\Desktop\OTClient
# Linux/Mac: /home/usuario/OTClient
CLIENT_PATH=C:/caminho/para/seu/cliente

# Número de uploads simultâneos (padrão: 5)
# Aumente se tiver boa conexão (ex: 10 ou 15)
UPLOAD_CONCURRENCY=5
